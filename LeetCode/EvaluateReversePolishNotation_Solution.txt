class Solution {
    public int evalRPN(String[] tokens) {
        Stack<Integer> st=new Stack<>();
        for(String s:tokens){
            if(s.equals("+")||s.equals("-")||s.equals("*")||s.equals("/")){
                int op2=st.pop();
                int op1=st.pop();
                int result=0;
                switch(s){
                    case "+":
                        result=op1+op2;
                        break;
                    case "-":
                        result=op1-op2;
                        break;
                    case "*":
                        result=op1*op2;
                        break;
                    case "/":
                        result=op1/op2;
                        break;        
                }
                st.push(result);
            }
            else{
                st.push(Integer.parseInt(s));
            }
        }
        return st.peek();
    }
}
