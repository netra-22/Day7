class Solution {
    public int[] deckRevealedIncreasing(int[] deck) {
        Arrays.sort(deck);
        Deque<Integer> d=new ArrayDeque<>();
        for(int i=deck.length-1;i>=0;i--){
            if(!d.isEmpty()){
                d.addFirst(d.removeLast());
            }
            d.addFirst(deck[i]);
        }
        int[] dec=new int[deck.length];
        int x=0;
        for(int i:d){
            dec[x++]=i;
        }
        return dec;
    }
}
